{{- /* Release number for a post. A release is the article, not the translation: the English and Uzbek versions of the same post carry the same number. Posts are grouped by translationKey across every language, groups are ordered by their earliest date, and the position in that order is the number. Ties are broken by translationKey so the order is stable between builds. Default: computed. Deleting a post shifts every later number — set `id:` in the front matter to pin one permanently; it wins over the computed value. */ -}} {{- $page := . -}} {{- $n := 0 -}} {{- with $page.Params.id -}} {{- $n = . -}} {{- else -}} {{- $earliest := dict -}} {{- range hugo.Sites -}} {{- range (where .RegularPages "Section" "posts") -}} {{- $k := .TranslationKey -}} {{- $d := .Date.UTC -}} {{- $known := index $earliest $k -}} {{- if or (not $known) ($d.Before $known) -}} {{- $earliest = merge $earliest (dict $k $d) -}} {{- end -}} {{- end -}} {{- end -}} {{- $rows := slice -}} {{- range $k, $d := $earliest -}} {{- $rows = $rows | append (dict "key" $k "sortOn" (printf "%s|%s" ($d.Format "2006-01-02T15:04:05Z") $k)) -}} {{- end -}} {{- range $i, $r := (sort $rows "sortOn" "asc") -}} {{- if eq $r.key $page.TranslationKey -}} {{- $n = add $i 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- printf "#%04d" $n -}}
